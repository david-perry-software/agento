# Step 4.1 dual-merge drive — 2026-09-17
# Script: /tmp/agento-ship-dual-merge-drive.sh (throwaway /tmp product+companion pair, bare origins; stub gh on PATH logs "$PWD $*" and answers from a state file; PRs #12 code / #7 companion)
# Invocation: AGENTO_CLI=<worktree>/scripts/agento.mjs AGENTO_WFC=<worktree>/scripts/wait-for-checks.sh bash /tmp/agento-ship-dual-merge-drive.sh
# Exit: 0 — run A (clean) and run B (companion pr merge fails once; re-send resumes at the companion merge) both assert OK
# Note: scripts/wait-for-checks.sh ran unmodified against the stub; the stub answers `pr view … --jq` with the snapshot line "1 0 0 0 <mergeStateStatus> | " directly (no jq evaluation), so each wait returned RESULT: success on poll 1.
# Note: `pr merge` in the stub performs a real --no-ff merge into the bare origin main and deletes the remote branch, so the default-branch syncs (fetch --prune, merge --ff-only) exercise real git state.

#!/usr/bin/env bash
# Step 4.1 drive: /agento ship dual merge on a throwaway product+companion pair.
# Runs the literal command sequence of .github/prompts/ship.prompt.md (companion mode,
# owner !== null path) twice: run A clean, run B with the companion `gh pr merge`
# failing once and the re-send resuming at the companion merge. `gh` is a stub on
# PATH that logs "$PWD $*" and answers from a state file; `pr merge` really merges
# the branch into the bare origin's main and deletes the remote branch.
set -euo pipefail
CLI="${AGENTO_CLI:?set AGENTO_CLI to scripts/agento.mjs}"
WFC="${AGENTO_WFC:?set AGENTO_WFC to scripts/wait-for-checks.sh}"
say() { printf '\n### %s\n' "$*"; }
show() { printf '$ %s\n' "$*"; "$@"; }
g() { git -C "$1" "${@:2}"; }
jsonq() { python3 -c 'import json,sys; r=json.load(sys.stdin); print(json.dumps({k:r.get(k) for k in sys.argv[1:]}, indent=1))' "$@"; }

write_stub_gh() {
  cat > "$1" <<'STUB'
#!/usr/bin/env bash
# Stub gh: logs "$PWD $*" and answers from $GH_STUB_STATE/<repo>.sh (repo = product | companion,
# chosen by --repo when given, else by $PWD containing project-docs).
set -uo pipefail
printf '%s %s\n' "$PWD" "$*" >> "$GH_STUB_LOG"
repo=product; [[ "$PWD" == *project-docs* ]] && repo=companion
args=("$@")
for ((i=0; i<${#args[@]}; i++)); do
  if [[ "${args[$i]}" == --repo ]]; then
    if [[ "${args[$((i+1))]}" == *project-docs* ]]; then repo=companion; else repo=product; fi
  fi
done
state="$GH_STUB_STATE/$repo.sh"
# shellcheck disable=SC1090
source "$state"
has() { local a; for a in "${args[@]}"; do [[ "$a" == "$1" ]] && return 0; done; return 1; }
setstate() { sed -i "s/^$1=.*/$1=$2/" "$state"; }
case "${1:-} ${2:-}" in
  "--version ") echo "gh version 0.0.0-stub"; exit 0 ;;
  "auth status") echo "Logged in to github.com as stub"; exit 0 ;;
  "repo view") echo "$nameWithOwner"; exit 0 ;;
  "pr view")
    if has --jq; then printf '1 0 0 0 %s | \n' "$mergeStateStatus"; exit 0; fi
    printf '{"number":%s,"state":"%s","isDraft":%s,"mergeStateStatus":"%s","mergeable":"MERGEABLE","url":"%s"}\n' \
      "$number" "$state_" "$isDraft" "$mergeStateStatus" "$url"; exit 0 ;;
  "pr checks") echo "no checks reported on pull request #$3"; exit 0 ;;
  "pr ready") setstate isDraft false; echo "✓ Pull request #$3 is marked as \"ready for review\""; exit 0 ;;
  "pr merge")
    if [[ -e "$GH_STUB_STATE/fail-$repo-merge-once" ]]; then
      rm -f "$GH_STUB_STATE/fail-$repo-merge-once"
      echo "X Pull request #$3 could not be merged: stub transient failure (HTTP 502)" >&2; exit 1
    fi
    origin=$(git remote get-url origin); tmp=$(mktemp -d)
    git clone -q "$origin" "$tmp"
    git -C "$tmp" -c user.name=gh-stub -c user.email=gh@stub merge -q --no-ff -m "Merge pull request #$number from $branch" "origin/$branch"
    git -C "$tmp" push -q origin main
    git -C "$tmp" push -q origin --delete "$branch"
    rm -rf "$tmp"
    setstate state_ MERGED; setstate isDraft false
    echo "✓ Merged pull request #$3 (merge commit); ✓ Deleted branch $branch"; exit 0 ;;
  *) echo "stub gh: unhandled: $*" >&2; exit 64 ;;
esac
STUB
  chmod +x "$1"
}

setup_pair() { # sets BASE PRODUCT DOCS PLAN HALF ART WS LOG STATE
  BASE=$(mktemp -d /tmp/agento-ship-drive-XXXXXX)
  mk() { # mk <name> [config-json]
    local origin="$BASE/$1.git" work="$BASE/$1"
    git init --bare -q -b main "$origin"
    git clone -q "$origin" "$work"
    g "$work" config user.email test@example.com
    g "$work" config user.name Test
    if [[ -n "${2:-}" ]]; then mkdir -p "$work/.github"; printf '%s\n' "$2" > "$work/.github/agento.json"; g "$work" add -A; fi
    g "$work" commit -q --allow-empty -m init
    g "$work" push -q -u origin main
  }
  say "Setup: bare origins + clones (product .github/agento.json → artifacts.repo.name = project-docs)"
  mk project '{"worktrees":{"dir":"../wt"},"artifacts":{"repo":{"name":"project-docs"}}}'
  mk project-docs
  PRODUCT="$BASE/project"; DOCS="$BASE/project-docs"
  mkdir -p "$BASE/wt" "$BASE/project-docs-worktrees" "$BASE/bin"
  ID=20260917-1; PLAN="$BASE/wt/plan-$ID"; HALF="$BASE/project-docs-worktrees/plan-$ID"; WS="$BASE/wt/plan-$ID.code-workspace"
  STATE="$BASE/gh-state"; LOG="$BASE/gh-calls.log"; mkdir -p "$STATE"; : > "$LOG"
  export GH_STUB_STATE="$STATE" GH_STUB_LOG="$LOG"
  write_stub_gh "$BASE/bin/gh"
  export PATH="$BASE/bin:$PATH"
  show command -v gh
  cat > "$STATE/product.sh" <<EOF
number=12
state_=OPEN
isDraft=true
mergeStateStatus=CLEAN
url=https://github.com/acme/project/pull/12
branch=feature/xw
nameWithOwner=acme/project
EOF
  cat > "$STATE/companion.sh" <<EOF
number=7
state_=OPEN
isDraft=true
mergeStateStatus=CLEAN
url=https://github.com/acme/project-docs/pull/7
branch=feature/xw
nameWithOwner=acme/project-docs
EOF
  node "$CLI" --root "$PRODUCT" config | grep -E '"root"|"artifactsRoot"|"name"|"dir"'

  say "Build pair on feature/xw (start-session build mode: worktree add --no-track -b in both halves) + workspace file"
  show g "$PRODUCT" worktree add -q --no-track -b feature/xw "$PLAN" origin/main
  show g "$DOCS" worktree add -q --no-track -b feature/xw "$HALF" origin/main
  printf '{"folders":[{"path":"%s"},{"path":"%s"}]}\n' "$PLAN" "$HALF" > "$WS"
  echo widget > "$PLAN/widget.txt"
  g "$PLAN" add widget.txt; g "$PLAN" commit -q -m "feat(xw): add widget (step 1.1)"
  show g "$PLAN" push -q -u origin feature/xw
  ART="$HALF/features/2026/09/xw"; mkdir -p "$ART"
  cat > "$ART/plan.md" <<'EOF'
# xw

## Acceptance checklist

- [x] widget.txt exists on the product branch
EOF
  cat > "$ART/roadmap.md" <<'EOF'
```yaml
status: in-review
branch: feature/xw
last-updated: 2026-09-17
next-step: ""
artifact-pr: "#7"
```

## Phase 1

- [x] 1.1 add widget.txt — verify: test -f widget.txt
EOF
  cat > "$ART/review.md" <<'EOF'
# Review: xw

Verdict: approve

## Acceptance checklist

- [x] widget.txt exists on the product branch — `test -f widget.txt` exit 0
EOF
  g "$HALF" add features; g "$HALF" commit -q -m "docs(feature): xw plan, roadmap in-review, review approve (artifact-pr #7)"
  show g "$HALF" push -q -u origin feature/xw
  echo "(code PR #12 on acme/project, companion PR #7 on acme/project-docs — stub state, no GitHub here)"
  show node "$CLI" --root "$PRODUCT" paths feature xw
}

preflight() { # preflight <stage>
  say "ship-preflight feature xw --pr — stage: $1"
  local rc=0 out
  out=$(cd "$PRODUCT" && node "$CLI" ship-preflight feature xw --pr) || rc=$?
  echo "exit $rc"
  jsonq status resolutionSource branch owner companion companionGaps pr companionPr warnings message <<<"$out"
}

audit_and_merge_code() {
  cd "$PRODUCT"
  say "find + doctor --for ship (stub gh answers --version / auth status)"
  show node "$CLI" find xw
  (node "$CLI" doctor --for ship || true) | grep -E '"status"|"id"' | sed -n '1,14p'
  preflight "audit (both PRs OPEN, draft)"

  say "Audit (read-only from the primary; artifacts from the companion's origin/<branch>)"
  show git fetch origin
  show g "$DOCS" fetch origin
  show g "$DOCS" show origin/feature/xw:features/2026/09/xw/roadmap.md
  show g "$DOCS" show origin/feature/xw:features/2026/09/xw/review.md
  show git diff --stat origin/main...origin/feature/xw
  printf '$ git -C PLAN status --porcelain → [%s]\n' "$(g "$PLAN" status --porcelain)"
  printf '$ git -C PLAN rev-list --count @{upstream}..HEAD → %s\n' "$(g "$PLAN" rev-list --count @{upstream}..HEAD)"
  show gh pr view 12 --json mergeStateStatus,mergeable
  (cd "$DOCS" && show gh pr view 7 --json mergeStateStatus,mergeable)
  show gh pr checks 12
  printf '$ git diff origin/main...origin/feature/xw -- .claude-plugin/plugin.json package.json → [%s] (no version change, no changelog stamp)\n' "$(git diff origin/main...origin/feature/xw -- .claude-plugin/plugin.json package.json)"

  say "Step 3: status: complete committed and pushed in the companion half (git -C <companion.path>)"
  sed -i 's/^status: in-review$/status: complete/' "$ART/roadmap.md"
  show g "$HALF" commit -q -am "docs(feature): ship xw — status complete"
  show g "$HALF" push -q origin feature/xw
  preflight "after status: complete push (companion ahead 0)"

  say "Step 3: mark both PRs ready; wait on the code PR's checks; merge the code PR"
  show gh pr ready 12
  (cd "$DOCS" && show gh pr ready 7)
  show bash "$WFC" pr 12 --interval 1 --max-seconds 10
  show gh pr merge 12 --merge --delete-branch

  say "Sync the product default in the primary"
  show git switch -q main
  show git fetch --prune -q
  show git merge --ff-only -q origin/main
  printf '$ git status --porcelain → [%s]; ahead/behind → %s\n' "$(git status --porcelain)" "$(git rev-list --left-right --count @{upstream}...HEAD | tr '\t' '/')"
  show git show origin/main:widget.txt
  preflight "code PR MERGED, companion PR OPEN"
}

merge_companion() {
  cd "$PRODUCT"
  say "Step 3 (companion mode): mark the companion PR ready, wait, merge — from inside the clone / --repo <nameWithOwner>"
  NWO=$(cd "$DOCS" && gh repo view --json nameWithOwner -q .nameWithOwner)
  echo "nameWithOwner (gh repo view inside the clone) = $NWO"
  (cd "$DOCS" && show gh pr ready 7)
  show bash "$WFC" pr 7 --repo "$NWO" --interval 1 --max-seconds 10
  if (cd "$DOCS" && show gh pr merge 7 --merge --delete-branch); then
    return 0
  fi
  echo "Result: failed — code PR #12 merged, companion PR #7 open at https://github.com/acme/project-docs/pull/7; re-send /agento ship xw to resume at the companion merge"
  return 1
}

sync_companion_and_teardown() {
  cd "$PRODUCT"
  say "Sync the companion default"
  show g "$DOCS" switch -q main
  show g "$DOCS" fetch --prune -q
  show g "$DOCS" merge --ff-only -q origin/main
  printf '$ git -C DOCS status --porcelain → [%s]; ahead/behind → %s\n' "$(g "$DOCS" status --porcelain)" "$(g "$DOCS" rev-list --left-right --count @{upstream}...HEAD | tr '\t' '/')"
  g "$DOCS" show origin/main:features/2026/09/xw/roadmap.md | sed -n '2p'
  preflight "both PRs MERGED, remote branches deleted, owner worktree still present"

  say "Teardown: worktree lists before"
  show git worktree list --porcelain
  show g "$DOCS" worktree list --porcelain
  show g "$DOCS" worktree remove "$HALF"
  show g "$DOCS" worktree prune
  show git worktree remove "$PLAN"
  show git worktree prune
  show git branch -d feature/xw
  show g "$DOCS" branch -d feature/xw
  show rm "$WS"
  say "Teardown: worktree lists after"
  show git worktree list --porcelain
  show g "$DOCS" worktree list --porcelain
  printf '$ ls wt → [%s]; ls project-docs-worktrees → [%s]\n' "$(ls "$BASE/wt")" "$(ls "$BASE/project-docs-worktrees")"

  say "Final state: both defaults clean, zero ahead/behind, both merges on both origins"
  printf 'product: status [%s] ahead/behind %s; branch feature/xw → %s\n' "$(git status --porcelain)" "$(git rev-list --left-right --count @{upstream}...HEAD | tr '\t' '/')" "$(git rev-parse --verify --quiet feature/xw || echo gone)"
  printf 'companion: status [%s] ahead/behind %s; branch feature/xw → %s\n' "$(g "$DOCS" status --porcelain)" "$(g "$DOCS" rev-list --left-right --count @{upstream}...HEAD | tr '\t' '/')" "$(g "$DOCS" rev-parse --verify --quiet feature/xw || echo gone)"
  show git log --oneline -3 origin/main
  show g "$DOCS" log --oneline -3 origin/main
  printf 'remote heads product: %s | companion: %s\n' "$(git ls-remote --heads origin | awk '{print $2}' | tr '\n' ' ')" "$(g "$DOCS" ls-remote --heads origin | awk '{print $2}' | tr '\n' ' ')"
}

check_log() { # check_log <run> <expect-marker 0|1>
  say "gh call log ($1)"
  cat "$LOG"
  python3 - "$LOG" "$BASE" "$2" <<'EOF'
import sys
log, base, expect_marker = open(sys.argv[1]).read().splitlines(), sys.argv[2], sys.argv[3] == "1"
docs = base + "/project-docs"
def pwd(l): return l.split(" ", 1)[0]
merges = [i for i, l in enumerate(log) if " pr merge " in l]
prod = [i for i in merges if " pr merge 12 " in log[i]]
comp = [i for i in merges if " pr merge 7 " in log[i]]
assert prod and comp and prod[0] < comp[0], (prod, comp)
for l in log:
    if " pr merge 12" in l or " pr ready 12" in l or " pr view 12" in l or " pr checks 12" in l:
        assert pwd(l) == base + "/project", l
    if " pr merge 7" in l or " pr ready 7" in l or " repo view" in l:
        assert pwd(l).startswith(docs), l
    if " pr view 7" in l:
        assert pwd(l).startswith(docs) or "--repo acme/project-docs" in l, l
if expect_marker:
    marker = log.index("--- re-send /agento ship xw: pass 2 ---")
    second = log[marker + 1:]
    assert not any(" pr merge 12" in l for l in second), second
    assert sum(" pr merge 7 " in l for l in log) == 2, log
    print(f"OK: run B — product merge at line {prod[0]+1}, failed companion merge at {comp[0]+1}, pass 2 starts at {marker+1}, second companion merge at {comp[1]+1}; no product pr merge in pass 2")
else:
    assert len(prod) == 1 and len(comp) == 1
    print(f"OK: run A — product pr merge at line {prod[0]+1} before companion pr merge at line {comp[0]+1}; every companion gh call ran with $PWD in the clone or --repo acme/project-docs")
EOF
}

# ------------------------------------------------------------------ run A
say "================ RUN A: clean dual merge ================"
setup_pair
audit_and_merge_code
merge_companion
sync_companion_and_teardown
check_log A 0
rm -rf "$BASE"; echo "cleaned $BASE"

# ------------------------------------------------------------------ run B
say "================ RUN B: companion merge fails once, re-send resumes at the companion merge ================"
setup_pair
touch "$STATE/fail-companion-merge-once"
audit_and_merge_code
if merge_companion; then echo "UNEXPECTED: companion merge succeeded on pass 1"; exit 1; fi
say "Half-shipped state as the re-send sees it"
preflight "re-send: pr MERGED / companionPr OPEN → skip audit writes and the code merge, resume at the companion merge"
echo '$ node CLI --root PLAN session --pr | warnings, lifecycle:'
(node "$CLI" --root "$PLAN" session --pr || true) | jsonq lifecycle warnings
echo "--- re-send /agento ship xw: pass 2 ---" >> "$LOG"
merge_companion
sync_companion_and_teardown
check_log B 1
rm -rf "$BASE"; echo "cleaned $BASE"
echo
echo "DRIVE OK"
